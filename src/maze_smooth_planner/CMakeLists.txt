cmake_minimum_required(VERSION 3.0.2)
project(maze_smooth_planner)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp nav_core global_planner costmap_2d pluginlib geometry_msgs nav_msgs std_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES maze_smooth_planner CATKIN_DEPENDS roscpp nav_core global_planner costmap_2d pluginlib geometry_msgs nav_msgs std_msgs)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(maze_smooth_planner src/smoother.cpp src/planner.cpp)
target_link_libraries(maze_smooth_planner ${catkin_LIBRARIES})
install(TARGETS maze_smooth_planner ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(FILES planner_plugin.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(smoother_geometry_test test/smoother_geometry_test.cpp)
  if(TARGET smoother_geometry_test)
    target_link_libraries(smoother_geometry_test maze_smooth_planner ${catkin_LIBRARIES})
  endif()
endif()
